{{- /* Overrides the theme's: a link to another page is written as the file (`handlers.md`), so it works on GitHub too, and a missing file fails the build (the theme would link to the file on GitHub). */ -}} {{- $url := urls.Parse .Destination -}} {{- $href := .Destination -}} {{- if and (not $url.IsAbs) (strings.HasSuffix $url.Path ".md") -}} {{- with .PageInner.GetPage $url.Path -}} {{- $href = .RelPermalink -}} {{- with $url.Fragment }}{{ $href = printf "%s#%s" $href . }}{{ end -}} {{- else -}} {{- errorf "%s: no page for link %q" .PageInner.File.Path .Destination -}} {{- end -}} {{- end -}} {{ .Text }}
